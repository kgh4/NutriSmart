💡 BEFORE vs AFTER

  BEFORE (Unsafe)                  AFTER (Safe)
  ──────────────────────────────────────────────────────────────────
  val name = user!!.name      →    val name = user?.name ?: "Unknown"
  val first = recipes[0]      →    val first = recipes.firstOrNull()
  val id = args?.getString()!!→   if (id?.isNotBlank() == true) { ... }
  riskyOp()                   →    try { riskyOp() } catch(e) { ... }
  Text(user?.name)            →    Text(user?.name ?: "Unknown")
